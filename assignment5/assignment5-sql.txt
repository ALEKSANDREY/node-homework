# Task 1: Total price of first 5 orders ordered by order_id
SELECT o.order_id, SUM(p.price * l.quantity) AS total_price
FROM (
    SELECT order_id FROM orders ORDER BY order_id LIMIT 5
) o
JOIN line_items l ON o.order_id = l.order_id
JOIN products p ON l.product_id = p.product_id
GROUP BY o.order_id
ORDER BY o.order_id;

# Task 2: Average total price of orders per customer
SELECT c.customer_name, AVG(t.order_total) AS average_order_price
FROM customers c
JOIN (
    SELECT o.order_id, o.customer_id, SUM(p.price * l.quantity) AS order_total
    FROM orders o
    JOIN line_items l ON o.order_id = l.order_id
    JOIN products p ON l.product_id = p.product_id
    GROUP BY o.order_id, o.customer_id
) t ON c.customer_id = t.customer_id
GROUP BY c.customer_id, c.customer_name
ORDER BY c.customer_name;

# Task 4: Employees with more than 5 orders
SELECT e.first_name, e.last_name, COUNT(o.order_id) AS order_count
FROM employees e
JOIN orders o ON e.employee_id = o.employee_id
GROUP BY e.employee_id, e.first_name, e.last_name
HAVING COUNT(o.order_id) > 5
ORDER BY e.last_name ASC;
